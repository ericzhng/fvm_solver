\section*{Derivation of Roe Flux for Euler Equations and Isentropic Gas System}

\subsection*{Euler Equations}
The 1D Euler equations for inviscid compressible flow are:
\[
\mathbf{u} = \begin{bmatrix} \rho \\ \rho u \\ \rho E \end{bmatrix}, \quad \mathbf{f} = \begin{bmatrix} \rho u \\ \rho u^2 + p \\ \rho u E + p u \end{bmatrix}
\]
where \(\rho\) is density, \(u\) is velocity, \(E\) is total energy per unit mass, \(p\) is pressure, and \(E = e + \frac{u^2}{2}\) with \(e = \frac{p}{(\gamma-1)\rho}\) for an ideal gas (\(\gamma\) is the adiabatic index).

\subsubsection*{Roe Flux Process}
\begin{enumerate}
    \item \textbf{Jacobian}: The flux Jacobian is:
    \[
    \mathbf{A} = \begin{bmatrix}
    0 & 1 & 0 \\
    -\frac{1}{2}(\gamma-3)u^2 & (3-\gamma)u & \gamma-1 \\
    -\frac{\gamma u E}{\gamma-1} + \frac{\gamma-1}{2}u^3 & \frac{\gamma E}{\gamma-1} - \frac{3-\gamma}{2}u^2 & \gamma u
    \end{bmatrix}
    \]
    Eigenvalues: \(\lambda_1 = u - c\), \(\lambda_2 = u\), \(\lambda_3 = u + c\), where \(c = \sqrt{\gamma p / \rho}\). Eigenvectors:
    \[
    \mathbf{r}_1 = \begin{bmatrix} 1 \\ u - c \\ H - u c \end{bmatrix}, \quad \mathbf{r}_2 = \begin{bmatrix} 1 \\ u \\ \frac{u^2}{2} \end{bmatrix}, \quad \mathbf{r}_3 = \begin{bmatrix} 1 \\ u + c \\ H + u c \end{bmatrix}
    \]
    where \(H = E + \frac{p}{\rho} = \frac{c^2}{\gamma-1} + \frac{u^2}{2}\).

    \item \textbf{Roe Averaging}: For left (\(\mathbf{u}_L = [\rho_L, \rho_L u_L, \rho_L E_L]^T\)) and right (\(\mathbf{u}_R\)) states:
    \[
    \tilde{\rho} = \sqrt{\rho_L \rho_R}, \quad \tilde{u} = \frac{\sqrt{\rho_L} u_L + \sqrt{\rho_R} u_R}{\sqrt{\rho_L} + \sqrt{\rho_R}}, \quad \tilde{H} = \frac{\sqrt{\rho_L} H_L + \sqrt{\rho_R} H_R}{\sqrt{\rho_L} + \sqrt{\rho_R}}
    \]
    Roe-averaged sound speed: \(\tilde{c} = \sqrt{(\gamma-1)\left(\tilde{H} - \frac{\tilde{u}^2}{2}\right)}\). Eigenvalues: \(\tilde{\lambda}_1 = \tilde{u} - \tilde{c}\), \(\tilde{\lambda}_2 = \tilde{u}\), \(\tilde{\lambda}_3 = \tilde{u} + \tilde{c}\).

    \item \textbf{Wave Strengths}: Solve \(\Delta \mathbf{u} = \mathbf{u}_R - \mathbf{u}_L = \sum_{k=1}^3 \tilde{\alpha}_k \tilde{\mathbf{r}}_k\):
    \[
    \tilde{\alpha}_1 = \frac{\Delta p + \tilde{\rho} \tilde{c} \Delta u}{2 \tilde{c}^2}, \quad \tilde{\alpha}_2 = \Delta \rho - \frac{\Delta p}{\tilde{c}^2}, \quad \tilde{\alpha}_3 = \frac{\Delta p - \tilde{\rho} \tilde{c} \Delta u}{2 \tilde{c}^2}
    \]
    where \(\Delta \rho = \rho_R - \rho_L\), \(\Delta u = u_R - u_L\), \(\Delta p = p_R - p_L\).

    \item \textbf{Roe Flux}:
    \[
    \mathbf{F}_{i+1/2} = \frac{1}{2} \left( \mathbf{f}(\mathbf{u}_L) + \mathbf{f}(\mathbf{u}_R) \right) - \frac{1}{2} \sum_{k=1}^3 |\tilde{\lambda}_k| \tilde{\alpha}_k \tilde{\mathbf{r}}_k
    \]
\end{enumerate}

\subsection*{Isentropic Gas System}
For isentropic flow (\(p = K \rho^\gamma\)), the equations are:
\[
\mathbf{u} = \begin{bmatrix} \rho \\ \rho u \end{bmatrix}, \quad \mathbf{f} = \begin{bmatrix} \rho u \\ \rho u^2 + \frac{K \gamma}{\gamma-1} \rho^{\gamma-1} \end{bmatrix}
\]

\subsubsection*{Roe Flux Process}
\begin{enumerate}
    \item \textbf{Jacobian}:
    \[
    \mathbf{A} = \begin{bmatrix} 0 & 1 \\ -u^2 + K \gamma \rho^{\gamma-1} & 2u \end{bmatrix}
    \]
    Eigenvalues: \(\lambda_1 = u - c\), \(\lambda_2 = u + c\), where \(c = \sqrt{K \gamma \rho^{\gamma-1}}\). Eigenvectors:
    \[
    \mathbf{r}_1 = \begin{bmatrix} 1 \\ u - c \end{bmatrix}, \quad \mathbf{r}_2 = \begin{bmatrix} 1 \\ u + c \end{bmatrix}
    \]

    \item \textbf{Roe Averaging}:
    \[
    \tilde{\rho} = \sqrt{\rho_L \rho_R}, \quad \tilde{u} = \frac{\sqrt{\rho_L} u_L + \sqrt{\rho_R} u_R}{\sqrt{\rho_L} + \sqrt{\rho_R}}, \quad \tilde{c} = \sqrt{K \gamma \tilde{\rho}^{\gamma-1}}
    \]
    Eigenvalues: \(\tilde{\lambda}_1 = \tilde{u} - \tilde{c}\), \(\tilde{\lambda}_2 = \tilde{u} + \tilde{c}\).

    \item \textbf{Wave Strengths}: Solve \(\Delta \mathbf{u} = \tilde{\alpha}_1 \tilde{\mathbf{r}}_1 + \tilde{\alpha}_2 \tilde{\mathbf{r}}_2\):
    \[
    \tilde{\alpha}_1 = \frac{\Delta (\rho u) - \tilde{u} \Delta \rho + \tilde{c} \Delta \rho}{2 \tilde{c}}, \quad \tilde{\alpha}_2 = \frac{\Delta (\rho u) - \tilde{u} \Delta \rho - \tilde{c} \Delta \rho}{-2 \tilde{c}}
    \]

    \item \textbf{Roe Flux}:
    \[
    \mathbf{F}_{i+1/2} = \frac{1}{2} \left( \mathbf{f}(\mathbf{u}_L) + \mathbf{f}(\mathbf{u}_R) \right) - \frac{1}{2} \sum_{k=1}^2 |\tilde{\lambda}_k| \tilde{\alpha}_k \tilde{\mathbf{r}}_k
    \]
\end{enumerate}

Both derivations ensure a conservative, consistent flux satisfying the Rankine-Hugoniot condition.
